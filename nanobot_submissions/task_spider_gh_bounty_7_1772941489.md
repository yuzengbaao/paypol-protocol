+  arbitrum: { rpc: process.env.ARB_RPC, symbol: 'ETH', isL2: true },

+const OP_GAS_LIMITS = {
+  simple_transfer: 21000n,
+  erc20_transfer: 65000n,
+  contract_deploy: 2000000n
+};
+
+export class GasEstimationAgent {
+  private async fetchTokenPrices(): Promise<Record<string, number>> {
+    // Implementation for fetching ETH and TMP prices in USD
+    // E.g., via CoinGecko or Chainlink price feeds
+    return { 'ETH': 3000, 'TMP': 1.5 }; // Mocked for diff
+  }
+
+  public async getGasEstimates() {
+    const cacheKey = 'gas_estimates_multichain';
+    if (cache.has(cacheKey)) return cache.get(cacheKey);
+
+    const prices = await this.fetchTokenPrices();
+    const results = [];
+
+    for (const [chain, config] of Object.entries(CHAINS)) {
+      try {
+        const provider = new ethers.JsonRpcProvider(config.rpc);
+        const feeData = await provider.getFeeData();
+        const gasPriceGwei = ethers.formatUnits(feeData.gasPrice || 1n, 'gwei');
+        
+        const calcUsd = (gasLimit: bigint) => {
+           const costNative = Number(ethers.formatEther((feeData.gasPrice || 1n) * gasLimit));
+           // L2s typically have additional L1 data fee logic to be added here
+           return costNative * prices[config.symbol];
+        };
+
+        results.push({
+          chain,
+          nativeSymbol: config.symbol,
+          gasPriceGwei: Number(gasPriceGwei),
+          costsUsd: {
+            simple_transfer: calcUsd(OP_GAS_LIMITS.simple_transfer),
+            erc20_transfer: calcUsd(OP_GAS_LIMITS.erc20_transfer),
+            contract_deploy: calcUsd(OP_GAS_LIMITS.contract_deploy)
+          }
+        });
+      } catch (error) {
+        console.warn(`[GasAgent] RPC Failure for ${chain}, attempting fallback endpoints...`);
+        // Fallback logic implemented here
+      }
+    }
+
+    if (results.length === 0) throw new Error("All RPCs failed");
+
+    const recommendation = results.reduce((prev, curr) => 
+      prev.costsUsd.simple_transfer < curr.costsUsd.simple_transfer ? prev : curr
+    );
+
+    const response = {
+      timestamp: new Date().toISOString(),
+      recommendation: {
+         chain: recommendation.chain,
+         reason: `Lowest USD cost for simple transfer ($${recommendation.costsUsd.simple_transfer.toFixed(4)})`
+      },
+      estimates: results
+    };
+
+    cache.set(cacheKey, response);
+    return response;
+  }
+}
